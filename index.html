📖 Начало работы с Arduino IDE

Полное руководство по скачиванию и установке Arduino IDE на Windows и Linux.

🔻 Шаг 1: Скачивание Arduino IDE

Официальный сайт (требует VPN):

🌐 https://www.arduino.cc/en/software

Альтернатива от Amperka (без VPN):

🌐 https://amperka.ru/page/arduino-ide

💡 Обе версии идентичны. Выберите что вам удобнее!

🪟 Установка на Windows

Вариант 1: Установка через установщик

  1. Скачайте файл ArduinoSetup.exe с сайта
  2. Дважды кликните по установщику
  3. Нажмите "I Agree" (согласиться с лицензией)
  4. Выберите компоненты для установки (оставьте всё по умолчанию)
  5. Нажмите "Next" несколько раз
  6. Выберите папку для установки (рекомендуется C:\Program Files\Arduino)
  7. Ждите завершения установки (может занять 5-10 минут)
  8. На рабочем столе появится ярлык Arduino IDE

Вариант 2: Портативная версия (без установки)

  1. Скачайте файл arduino-ide_windows.zip
  2. Откройте Проводник (File Explorer)
  3. Правой кнопкой мыши кликните на архив → "Извлечь всё"
  4. Выберите папку (например, C:\Arduino IDE)
  5. Запустите файл arduino-ide.exe

⚠️ Важно для Windows:

  • Если Arduino не узнаёт ваш COM-порт — скачайте драйверы USB FTDI или CH340
  • В Диспетчере устройств может появиться "Unknown Device" — это нормально, нужны драйверы
  • После установки драйверов перезагрузите компьютер

🐧 Установка на Linux

Ubuntu/Debian (самый простой способ):

Откройте Терминал (Ctrl+Alt+T) и выполните команду:

sudo apt update && sudo apt install arduino -y

Или установите через snap:

sudo snap install arduino

Fedora/CentOS/RHEL:

sudo dnf install arduino -y

Arch Linux:

sudo pacman -S arduino -y

Ручная установка (через архив):

  1. Скачайте arduino-ide_linux64.tar.xz
  2. Откройте Терминал в папке со скачиванием
  3. Выполните команду распаковки:
tar -xf arduino-ide_linux64.tar.xz -C ~/
  1. Запустите Arduino IDE:
~/arduino-ide/arduino-ide

⚠️ Важно для Linux:

  • Для доступа к USB-портам, добавьте пользователя в группу dialout:
sudo usermod -a -G dialout $USER
  • После этого перезагрузитесь или выполните: newgrp dialout

✅ Проверка установки

Запустите Arduino IDE и проверьте:

  1. Откройте Файл → Примеры → 01. Basics → Blink
  2. Подключите Arduino к компьютеру через USB
  3. Выберите тип платы: Инструменты → Плата → Arduino Uno (или ваша плата)
  4. Выберите COM-порт: Инструменты → Порт → COM3 (или другой)
  5. Нажмите кнопку Загрузить (стрелка вправо)
  6. Если светодиод на плате начал мигать — всё работает! ✨

// ID разрешенных пользователей
const String allowedUsers[] = {
  "1234567890",
  "123456789",
  "1234567890"
};
const int totalUsers = 3;
// =================================

#define DHTPIN 2
#define DHTTYPE DHT22

DHT dht(DHTPIN, DHTTYPE);
WiFiClientSecure client;
UniversalTelegramBot bot(BOT_TOKEN, client);

float temperature = 0;
float humidity = 0;
unsigned long lastTime = 0;

bool isAllowed(String chat_id) {
  for(int i = 0; i < totalUsers; i++) {
    if(allowedUsers[i] == chat_id) return true;
  }
  return false;
}

void setup() {
  Serial.begin(115200);
  dht.begin();
  client.setInsecure();
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) delay(500);
  bot.sendMessage("404415869", "꧁༺ *БОТ ЗАПУЩЕН* ༻꧂", "Markdown");
  bot.sendMessage("332677564", "꧁༺ *БОТ ЗАПУЩЕН* ༻꧂", "Markdown");
  bot.sendMessage("8473046860", "꧁༺ *БОТ ЗАПУЩЕН* ༻꧂", "Markdown");
}

void loop() {
  if(millis() - lastTime > 5000) {
    float t = dht.readTemperature();
    float h = dht.readHumidity();
    if(!isnan(t) && !isnan(h)) { temperature = t; humidity = h; }
    
    int msgCount = bot.getUpdates(bot.last_message_received + 1);
    for(int i = 0; i < msgCount; i++) {
      String chat = bot.messages[i].chat_id;
      String text = bot.messages[i].text;
      String name = bot.messages[i].from_name;
      
      if(!isAllowed(chat)) { bot.sendMessage(chat, "⛔️ *НЕТ ДОСТУПА*", "Markdown"); continue; }

📟 КОД: ИЗМЕРЕНИЕ РАССТОЯНИЯ HC-SR04 НА LCD

#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// Настройка LCD
LiquidCrystal_I2C lcd(0x27, 16, 2);

// Пины для HC-SR04
#define TRIG_PIN 9
#define ECHO_PIN 10

void setup() {
  // Инициализация LCD
  lcd.init();
  lcd.backlight();
  
  // Настройка пинов HC-SR04
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  
  // Очистка экрана и вывод заголовка
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Дальномер");
  lcd.setCursor(0, 1);
  lcd.print("Готов к работе");
  delay(2000);
}

void loop() {
  // Генерируем ультразвуковой импульс
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  
  // Измеряем время Echo сигнала
  long duration = pulseIn(ECHO_PIN, HIGH);
  
  // Рассчитываем расстояние в сантиметрах
  // Звук проходит туда и обратно, поэтому делим на 2
  // Скорость звука ~343 м/с = 0.0343 см/мкс
  float distance = duration * 0.0343 / 2;
  
  // Выводим результат на LCD
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Расстояние:");
  
  // Форматируем вывод с 1 знаком после запятой
  lcd.setCursor(0, 1);
  if (distance >= 400 || distance <= 2) {
    lcd.print("Out of range");
  } else {
    lcd.print(distance);
    lcd.print(" cm");
    
    // Добавляем визуальную индикацию
    if (distance < 10) {
      lcd.setCursor(13, 1);
      lcd.print("!!");
    } else if (distance < 30) {
      lcd.setCursor(13, 1);
      lcd.print("!");
    }
  }
  
  // Небольшая задержка перед следующим измерением
  delay(500);
}

📟 КОД: БОМБА ИЗ КС2 С КЛАВИАТУРОЙ И LCD

#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <LedControl.h>
#include <Keypad.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);

// --- НАСТРОЙКИ МАТРИЦЫ 8х8 ---
// DIN = 9, CLK = 11, CS = 10
LedControl lc = LedControl(9, 11, 10, 1);

// --- ПИНЫ ЗУММЕРА И СВЕТОДИОДА ---
const int buzzerPin = 12; 
const int ledPin = 13;    // Внешний светодиод из CS2

// --- НАСТРОЙКИ КНОПОЧНОЙ МАТРИЦЫ ---
const byte ROWS = 4; 
const byte COLS = 4; 
char keys[ROWS][COLS] = {
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};
byte rowPins[ROWS] = {7, 6, 5, 4};     
byte colPins[COLS] = {3, 2, A0, A1};   
Keypad keypad = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);

// --- ЛОГИКА БОМБЫ ---
String plantCode = "7355608"; 
String defuseCode = "";        
String inputCode = "";         
bool isPlanted = false;        
bool isDefusing = false;       

int countdown = 40;          
unsigned long previousMillis = 0;
unsigned long beepMillis = 0;
int beepInterval = 1000;     

// Рисунки для матрицы 8х8
const byte skull[8] = { B00111100, B01011010, B11111111, B11011011, B11111111, B01111110, B01011010, B00111100 };
const byte c4_logo[8] = { B11111111, B10000001, B10111101, B10100101, B10100101, B10111101, B10000001, B11111111 };
const byte defused_icon[8] = { B00000000, B00000001, B00000011, B00000110, B01001100, B11011000, B01110000, B00100000 };

void setup() {
  lcd.init();
  lcd.backlight();
  lc.shutdown(0, false);       
  lc.setIntensity(0, 5);       
  lc.clearDisplay(0);          
  
  pinMode(buzzerPin, OUTPUT);
  pinMode(ledPin, OUTPUT); // Настраиваем пин светодиода на выход

  randomSeed(analogRead(A2)); 

  showIdleScreen();
}

void loop() {
  char key = keypad.getKey();
  unsigned long currentMillis = millis();

  // 1. РЕЖИМ ОЖИДАНИЯ (ЗАКЛАДКА)
  if (!isPlanted) {
    if (key) {
      if (key >= '0' && key <= '9') {
        if (inputCode.length() < 7) {
          inputCode += key;
          tone(buzzerPin, 1500, 50);
          updateInputScreen();
        }
      } else if (key == '*') {
        inputCode = "";
        tone(buzzerPin, 800, 150);
        updateInputScreen();
      } else if (key == '#') {
        if (inputCode == plantCode) {
          plantBomb();
        } else {
          lcd.clear();
          lcd.setCursor(3, 0);
          lcd.print("WRONG CODE!");
          tone(buzzerPin, 500, 500);
          delay(1500);
          inputCode = "";
          showIdleScreen();
        }
      }
    }
  } 
  
  // 2. БОМБА ТИКАЕТ
  else {
    // Каждую секунду уменьшаем таймер
    if (currentMillis - previousMillis >= 1000) {
      previousMillis = currentMillis;
      countdown--;
      
      // Ускорение темпа игры
      if (countdown > 20) beepInterval = 1000;
      else if (countdown > 10) beepInterval = 500;
      else if (countdown > 5) beepInterval = 250;
      else beepInterval = 120;
    }

    // Синхронное пиканье зуммера и мигание светодиода + матрицы
    if (currentMillis - beepMillis >= beepInterval) {
      beepMillis = currentMillis;
      
      // Включаем звук и зажигаем светодиод на 13 пине
      tone(buzzerPin, 2000, 60);
      digitalWrite(ledPin, HIGH); 
      
      static bool matrixOn = false;
      matrixOn = !matrixOn;
      if (matrixOn) drawMatrix(c4_logo);
      else lc.clearDisplay(0);
    }

    // Гасим светодиод чуть позже (через 60 мс после включения), чтобы он именно мигал, а не горел постоянно
    if (currentMillis - beepMillis >= 60) {
      digitalWrite(ledPin, LOW);
    }

    // Обработка дефуза
    if (key) {
      if (key == '*') {
        isDefusing = true;
        inputCode = "";
        generateDefuseCode();
        tone(buzzerPin, 1200, 100);
      } 
      else if (isDefusing && key >= '0' && key <= '9') {
        if (inputCode.length() < 4) {
          inputCode += key;
          tone(buzzerPin, 1500, 50);
        }
      } 
      else if (isDefusing && key == '#') {
        if (inputCode == defuseCode) {
          defuseBomb();
        } else {
          isDefusing = false;
          inputCode = "";
          tone(buzzerPin, 400, 400);
        }
      }
    }

    // Отрисовка экранов
    if (!isDefusing) {
      lcd.setCursor(0, 0);
      lcd.print("DETONATION IN:  ");
      lcd.setCursor(5, 1);
      if (countdown < 10) lcd.print("0");
      lcd.print(countdown);
      lcd.print(" SEC ");
    } else {
      lcd.setCursor(0, 0);
      lcd.print("DEFUSE CODE:");
      lcd.print(defuseCode);
      lcd.setCursor(0, 1);
      lcd.print("INPUT: [");
      lcd.print(inputCode);
      for(int i = inputCode.length(); i < 4; i++) {
        lcd.print("_");
      }
      lcd.print("] t-");
      if (countdown < 10) lcd.print("0");
      lcd.print(countdown);
    }

    if (countdown <= 0) {
      explode();
    }
  }
}

void showIdleScreen() {
  lcd.clear();
  lcd.setCursor(2, 0);
  lcd.print("ENTER C4 CODE:");
  lcd.setCursor(4, 1);
  lcd.print("[_______]");
}

void updateInputScreen() {
  lcd.setCursor(4, 1);
  lcd.print("[");
  lcd.print(inputCode);
  for (int i = inputCode.length(); i < 7; i++) {
    lcd.print("_");
  }
  lcd.print("]");
}

void plantBomb() {
  isPlanted = true;
  isDefusing = false;
  inputCode = "";
  countdown = 40;
  previousMillis = millis();
  
  lcd.clear();
  lcd.setCursor(2, 0);
  lcd.print("BOMB PLANTED");
  tone(buzzerPin, 2500, 800);
  digitalWrite(ledPin, HIGH); // Короткая вспышка при установке
  delay(1500);
  digitalWrite(ledPin, LOW);
  lcd.clear();
}

void generateDefuseCode() {
  defuseCode = "";
  for (int i = 0; i < 4; i++) {
    defuseCode += String(random(0, 10)); 
  }
}

void defuseBomb() {
  digitalWrite(ledPin, LOW); // Выключаем мигание тревоги
  lcd.clear();
  lcd.setCursor(1, 0);
  lcd.print("BOMB DEFUSED! ");
  lcd.setCursor(0, 1);
  lcd.print("COUNTER-TERRORIST");
  
  drawMatrix(defused_icon);
  
  tone(buzzerPin, 1000, 200); delay(200);
  tone(buzzerPin, 1300, 200); delay(200);
  tone(buzzerPin, 1700, 500);
  
  delay(5000); // Пауза перед новым раундом
  resetGame();
}

void explode() {
  lcd.clear();
  lcd.setCursor(3, 0);
  lcd.print("TERRORISTS");
  lcd.setCursor(6, 1);
  lcd.print("WIN!");
  drawMatrix(skull);

  // Светодиод горит постоянно во время взрыва
  digitalWrite(ledPin, HIGH); 

  for (int i = 0; i < 600; i++) {
    digitalWrite(buzzerPin, HIGH); delayMicroseconds(random(100, 1200));
    digitalWrite(buzzerPin, LOW); delayMicroseconds(random(100, 1200));
  }
  
  digitalWrite(ledPin, LOW);
  delay(5000); // Пауза перед новым раундом
  resetGame();
}

void resetGame() {
  isPlanted = false;
  isDefusing = false;
  inputCode = "";
  defuseCode = "";
  countdown = 40;
  digitalWrite(ledPin, LOW);
  lc.clearDisplay(0); 
  showIdleScreen();   
}

void drawMatrix(const byte image[8]) {
  for (int i = 0; i < 8; i++) {
    lc.setRow(0, i, image[i]);
  }
}
🧪 Датчики — подключение и примеры

Подборка популярных датчиков с проводкой, рекомендованными библиотеками и минимальными скетчами.

DHT22 / DHT11

Wiring: VCC → 5V/3.3V, GND → GND, DATA → цифровой пин (например D2). Резистор 10k между VCC и DATA.

Описание: Этот модуль измеряет температуру и относительную влажность. DHT11 дешёвый, но имеет ограниченную точность и интервал обновления 1 секунда. DHT22 точнее, поддерживает более широкий диапазон и подходит для домашних метеостанций и теплиц.

Library: DHT sensor library (Adafruit)

#include <DHT.h>
#define DHTPIN 2
#define DHTTYPE DHT22
DHT dht(DHTPIN, DHTTYPE);
void setup(){ Serial.begin(9600); dht.begin(); }
void loop(){ Serial.println(dht.readTemperature()); Serial.println(dht.readHumidity()); delay(2000); }

DS18B20

Wiring: VCC → 3.3/5V, GND → GND, DATA → пин (с резистором 4.7k к VCC). Libraries: OneWire + DallasTemperature.

Описание: Цифровой датчик температуры на одной проводной шине. Он удобен для измерений в воде и на поверхности, поддерживает до 8 датчиков на одной линии и легко масштабируется для распределённого мониторинга.

#include <OneWire.h>
#include <DallasTemperature.h>
#define ONE_WIRE_BUS 2
OneWire oneWire(ONE_WIRE_BUS);
DallasTemperature sensors(&oneWire);
void setup(){ Serial.begin(9600); sensors.begin(); }
void loop(){ sensors.requestTemperatures(); Serial.println(sensors.getTempCByIndex(0)); delay(1000); }

BME280 / BMP280

Wiring: I2C (SDA → A4, SCL → A5 на Uno; на ESP: SDA/D2, SCL/D1). Library: Adafruit_BME280.

Описание: BMP280 измеряет температуру и давление. BME280 добавляет влажность, поэтому это универсальный датчик для метеостанций, прогноза погоды и высотомеров в домашних проектах.

#include <Wire.h>
#include <Adafruit_BME280.h>
Adafruit_BME280 bme;
void setup(){ Serial.begin(9600); bme.begin(0x76); }
void loop(){ Serial.println(bme.readTemperature()); Serial.println(bme.readHumidity()); Serial.println(bme.readPressure()); delay(2000); }

HC-SR04 (ультразвук)

Wiring: VCC → 5V, GND → GND, TRIG → пин, ECHO → пин (используйте уровень защиты при 3.3V MCU).

Описание: Ультразвуковой дальномер измеряет расстояние по времени отклика звукового импульса. Подходит для роботов, автоматического парктроника и контроля заполнения ёмкостей, но не работает с мягкими и узкими объектами.

#define TRIG 9
#define ECHO 8
void setup(){ Serial.begin(9600); pinMode(TRIG, OUTPUT); pinMode(ECHO, INPUT); }
void loop(){ digitalWrite(TRIG, LOW); delayMicroseconds(2); digitalWrite(TRIG, HIGH); delayMicroseconds(10); digitalWrite(TRIG, LOW);
long dur = pulseIn(ECHO, HIGH); Serial.println(dur/58.0); delay(200); }

MQ-2 / MQ-серия

Wiring: VCC → 5V, GND → GND, AOUT → аналоговый пин. Требует калибровки; используйте аналоговое значение для порогов.

Описание: Газовые сенсоры на основе металлооксидов. MQ-2 подходит для дыма, пропана, бутана и метана. Они дешевы, но требуют прогрева и бережной настройки, поэтому лучше использовать в пороговых охранных системах, а не в качестве точного газоанализа.

int sensor = A0;
void setup(){ Serial.begin(9600); }
void loop(){ Serial.println(analogRead(sensor)); delay(500); }

PIR (датчик движения)

Wiring: VCC → 5V, GND → GND, OUT → цифровой пин.

Описание: Пассивный инфракрасный датчик фиксирует движение тепловых объектов. Используется в охранных системах, автоматическом освещении и энергосбережении. Работает стабильно при установке на стену или потолок.

int pir = 7;
void setup(){ Serial.begin(9600); pinMode(pir, INPUT); }
void loop(){ if(digitalRead(pir)) Serial.println("Motion"); delay(200); }

BH1750 (освещённость)

Wiring: I2C. Library: BH1750.

Описание: Цифровой люксметр измеряет освещённость в люксах и даёт стабильный результат, гораздо лучше чем дешевый LDR. Идеален для систем автоматического освещения и мониторинга дневного освещения.

#include <Wire.h>
#include <BH1750.h>
BH1750 lightMeter;
void setup(){ Serial.begin(9600); Wire.begin(); lightMeter.begin(); }
void loop(){ Serial.println(lightMeter.readLightLevel()); delay(1000); }

MPU6050 (акселерометр + гироскоп)

Wiring: I2C. Library: MPU6050 или Wire + DMP.

Описание: 6-осевой IMU измеряет ускорение и угловые скорости. Применяется в стабилизации дронов, детекции жестов и измерении наклона. Требует калибровки и фильтрации для точных данных.

#include <Wire.h>
#define MPU_ADDR 0x68
void setup(){ Wire.begin(); Serial.begin(9600); Wire.beginTransmission(MPU_ADDR); Wire.write(0x6B); Wire.write(0); Wire.endTransmission(true); }
void loop(){ Wire.beginTransmission(MPU_ADDR); Wire.write(0x3B); Wire.endTransmission(false); Wire.requestFrom(MPU_ADDR,6,true);
int16_t ax=Wire.read()<<8|Wire.read(); Serial.println(ax); delay(200); }

HX711 + тензодатчик (весы)

Wiring: HX711 DT → цифровой пин, SCK → цифровой пин. Library: HX711.

Описание: Усилитель для тензодатчика позволяет измерять силу и вес с хорошей точностью. Этот модуль используется в кухонных и промышленных весах, требует тарировки и калибровки, но даёт стабильные результаты при правильной настройке.

Soil moisture (резистивный/емкостной)

Wiring: аналоговый пин → датчик; для резистивного — осторожно с коррозией, лучше использовать емкостной датчик.

Описание: Резистивные датчики дешевы, но быстро ржавеют в земле. Емкостные работают дольше и точнее, идеально подходят для автополива и контроля влажности почвы.

🔥 Турбозажигалка

Описание: Мощная турбозажигалка с ветрозащитой — настоящая находка для радиолюбителя и мастера. Отлично подходит для усадки термоусадочных трубок при пайке проводов, обжигания оплётки, удаления заусенцев на пластике.
